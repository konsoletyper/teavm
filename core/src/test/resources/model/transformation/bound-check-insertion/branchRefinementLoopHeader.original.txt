var @this as this

$start
    @size := invokeStatic `Cond.size()I`
    @array := newArray I[@size]
    @i := invokeStatic `Cond.index()I`
    @zero := 0
    @c := @i compareTo @zero as int
    if @c >= 0 then goto $loop else goto $exit
$loop
    @v := @array[@i] as int
    invokeStatic `Cond.consume(I)V` @v
    @f := invokeStatic `Cond.invoke()I`
    if @f == 0 then goto $exit else goto $loop
$exit
    return @zero
